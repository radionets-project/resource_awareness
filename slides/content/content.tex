\begin{frame}{CodeCarbon: Practical Considerations}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      Good for:
      \vspace{1em}
      \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item Comparing model architectures
        \item Hyperparameter optimization experiments
        \item Single-machine or single-node workloads
        \item Quick integration into existing pipelines
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      Struggles with:
      \vspace{1em}
      \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item Ground-truth validation
        \item Multi-node HPC jobs
        \item Accuracy: $\pm10-20\%$ for GPU workloads
        \item Average regional CO₂ factors
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

        # Your resource-heavy code here

    # Or manually
    tracker = EnergyTracker(project_name="your_research_project")

\begin{darkframe}[t,fragile]{Example: Logging with MLFlow and LET}
  \begin{minted}{python}
    import mlflow
    import pandas as pd
    from lamarr_energy_tracker import EnergyTracker

  \end{minted}

  \begin{onlyenv}<2->
  \begin{minted}{python}
    with EnergyTracker(project_name="deep_learning", output_dir="/model/save/path") as tracker:
        # Your training/inference here
        mlflow.log_metric(...)  # e.g., loss
        mlflow.log_param(...)  # e.g., metadata (trainable params, batch size, architecture)
        pass

  \end{minted}
  \end{onlyenv}

  \begin{onlyenv}<3->
  \begin{minted}{python}
    # After training/inference
    emission_data = pd.read_csv("/model/save/path/emissions.csv").to_dict()

  \end{minted}
  \end{onlyenv}

  \begin{onlyenv}<4>
  \begin{minted}{python}
    eval_res = dict(
        running_time_total=emission_data["duration"][0],
        power_draw_total=emission_data["energy_consumed"][0] * 3.6e6,
        ...
    )

    for key, val in eval_res.items():
        mlflow.log_metric(key=key, value=value)
  \end{minted}
  \end{onlyenv}
\end{darkframe}

\begin{darkframe}[fragile]{STREP: Sustainable and Trustworthy Reporting for ML and AI}
  \begin{columns}[t]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{1em}
        \item Available on PyPI:
          \begin{minted}{shell-session}
              $ uv pip install strep[frontend]
          \end{minted}
        \item Visualise your data with a single script and a \texttt{JSON} file for metadata
      \end{itemize}
      \vspace{3em}
      \begin{minted}{python}
        from strep.index_scale import load_database, scale_and_rate
        from strep.elex.app import Visualization

        fname = 'path/to/your/database.pkl' # or .csv summary
        # load database and meta information (if available)
        database, meta = load_database(fname)
        # index-scale and rate database
        rated_database = scale_and_rate(database, meta)
        # start the interactive exploration tool
        app = Visualization(rated_database)
        app.run()
      \end{minted}
    \end{column}
    \hfill
    \begin{column}{0.38\textwidth}
      \texttt{meta\_properties.json}
      \begin{minted}{json}
        {
            "num_trainable_parameters": {
                "name": "Parameters",
                "shortname": "PAR",
                "unit": "Number",
                "group": "Resources"
            },
            "power_draw": {
                "name": "Power Draw per Epoch",
                "shortname": "ENE",
                "unit": "Wattseconds",
                "group": "Resources"
            }
        }
      \end{minted}
    \end{column}
  \end{columns}
\end{darkframe}
