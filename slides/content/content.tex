\begin{frame}{Static Estimation}
  
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item Statistical estimation
        \item Based on configuration and environment
        \item CO$_2$ Equiv. = Power ~\cdot~ Time ~\cdot~ CO$_2$ Efficiency
        \item ML Impact Calculator\footnote{\url{https://www.climatechange.ai/papers/neurips2019/22}}
        \item No flexible adjustments
      \end{itemize}
      \vspace*{1cm}
    \end{column}

\end{frame}

\begin{frame}{CodeCarbon: Overview}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item CO$_2$ emission tracking
        \item Developed by open-source community
        \item CPU, GPU and TPU workloads
        \item Real-time monitoring and post-run analysis
        \item Multi-GPU support
        \item $< 1\%$ performance overhead
      \end{itemize}
      \vspace*{1cm}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \includegraphics[height=4cm]{images/codecarbon_logo.png}\\

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{CodeCarbon: Practical Considerations}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      Good for:
      \vspace{1em}
      \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item Comparing model architectures
        \item Hyperparameter optimization experiments
        \item Single-machine or single-node workloads
        \item Quick integration into existing pipelines
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      Struggles with:
      \vspace{1em}
      \begin{itemize}
        \setlength{\itemsep}{1.5em}
        \item Ground-truth validation
        \item Multi-node HPC jobs
        \item Accuracy: $\pm10-20\%$ for GPU workloads
        \item Average regional CO₂ factors
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
